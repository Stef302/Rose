# ══════════════════ notes de types et coque de vérification ══════════════════
# L'émission RIR a besoin du type que le vérificateur a calculé pour chaque
# nœud (suffixes add.f, cst.q, destination de index…) et du type de lecture
# de chaque « lire » (message du dump). check_expr est la COQUE de ce_raw :
# elle note le type (index du nœud vers « base|elt ») à chaque visite saine
# puis le rend. Les secondes visites (MIXTE/NUM_MAX recalent l'argument)
# recréent des entrées identiques : la recherche lit la première, le doublon
# est inoffensif. Les notes voyagent en listes plates — une table en
# paramètre serait hors de portée de l'AOT — et l'ordre post-ordre des deux
# passes rend la recherche linéaire quasi immédiate.

fn check_expr($i: entier,
              $k: liste<chaine>, $tx: liste<chaine>, $tf: liste<chaine>,
              $a: liste<entier>, $b: liste<entier>, $as: liste<entier>, $ac: liste<entier>,
              $lg: liste<entier>, $al: liste<entier>,
              $vn: liste<chaine>, $vt: liste<chaine>, $gn: liste<chaine>, $gt: liste<chaine>,
              $fnoms: liste<chaine>, $fparams: liste<chaine>, $fret: liste<chaine>,
              $foi: liste<entier>, $fot: liste<chaine>, $fof: liste<chaine>,
              $err: liste<chaine>, $tni: liste<entier>, $tnt: liste<chaine>,
              $tri: liste<entier>, $trt: liste<chaine>) -> chaine {
  $t = ce_raw($i, $k, $tx, $tf, $a, $b, $as, $ac, $lg, $al,
              $vn, $vt, $gn, $gt, $fnoms, $fparams, $fret, $foi, $fot, $fof, $err,
              $tni, $tnt, $tri, $trt)
  si (longueur($err) == 1) {
    joindre($tni, $i)
    joindre($tnt, $t)
  }
  retour $t
}

# type d'une entrée de repli d'après son drapeau : « 1 » flottant, « q »
# rationnel, vide entier — sans jamais ratisser le texte (un pli comme
# 2^30000000 fait neuf millions de caractères que contient_barre ne doit
# pas parcourir, et le drapeau porte déjà l'information, comme le nœud
# littéral reconstitué côté C++ : T_ENTIER / T_RATIONNEL / T_FLOTTANT).
fn t_entree($f: chaine) -> chaine {
  si ($f == "1") { retour "flottant|vide" }
  si ($f == "q") { retour "rationnel|vide" }
  retour "entier|vide"
}

# dénominateur réduit à 1 ? — miroir fidèle de la normalisation de rnorm,
# calculé sur les entiers (le pgcd d'un géant et de 1 est un seul modulo).
fn rden1($p: entier, $q: entier) -> booleen {
  $np = $p
  $nq = $q
  si ($nq < 0) { $np = 0 - $np; $nq = 0 - $nq }
  si ($np == 0) { retour vrai }
  $g = pgcd2($np, $nq)
  si ($g > 1) { $nq = $nq div $g }
  retour $nq == 1
}

fn contient($s: chaine, $d: chaine) -> booleen {
  $n = longueur($s)
  $m = longueur($d)
  si ($m == 0) { retour vrai }
  $i = 0
  tant_que ($i + $m <= $n) {
    si (sous_chaine($s, $i, $i + $m) == $d) { retour vrai }
    $i = $i + 1
  }
  retour faux
}

# texte de littéral -> double au sens de lit_valeur_d : « p/q » vaut p divisé
# par q (le moteur fait strtod puis division), le reste passe par flottant(),
# le strtod du moteur — arrondi au plus proche identique.
fn val_d($v: chaine) -> flottant {
  $t = vue_texte($v)
  $n = longueur($t)
  $j = 0
  $ok = 1
  tant_que ($ok == 1) {
    si ($j >= $n) { $ok = 0 }
    sinon si (ordre($t[$j]) == 47) { $ok = 0 }
    sinon { $j = $j + 1 }
  }
  si ($j >= $n) { retour flottant($t) }
  retour flottant(sous_chaine($t, 0, $j)) / flottant(sous_chaine($t, $j + 1, $n))
}

# texte que le moteur écrit dans le RIR pour un flottant replié :
# texte_flottant = représentation la plus courte qui se relit à l'identique,
# « .0 » ajouté si elle n'a ni point ni exposant. chaine() imite ce texte
# SAUF pour les flottants entiers (|v| < 1e16) où il tronque via %lld :
# « 5.0 » devient « 5 » et « -0.0 » devient « 0 », signe perdu. On recolle
# donc « .0 » dans ce cas ; le signe du zéro se calcule sur place dans le
# pli (négation qui écrase un zéro, produit de facteurs de signes
# contraires) et se fige dans le texte stocké — 1.0 / 0.0 lève « division
# par zéro » dans le moteur, aucun sondage n'est possible, et les fonctions
# ne voyant pas les globales, rien d'autre ne peut charrier ce signe.
fn flot_txt($v: flottant, $zneg: booleen) -> chaine {
  $rs = chaine($v)
  $af = $v
  si ($af < 0.0) { $af = -$af }
  si ($af < 10000000000000000.0) {
    si ($v == entier($v)) {
      si ($v == 0.0) {
        $rs = "0.0"
        si ($zneg) { $rs = "-0.0" }
      }
      sinon {
        # flottant entier : chaine() passe par %lld — il faut le %.Ng de
        # texte_flottant. Chiffres significatifs = digits sans les zéros
        # finaux ; exposant = longueur - 1. printf : style fixe si
        # exposant < chiffres, sinon scientifique « d.ddde±dd ».
        $dg = chaine(entier($af))
        $e = longueur($dg) - 1
        $m = $dg
        $ok = 1
        tant_que ($ok == 1) {
          $nn = longueur($m)
          si ($nn <= 1) { $ok = 0 }
          sinon si (ordre($m[$nn - 1]) == 48) { $m = sous_chaine($m, 0, $nn - 1) }
          sinon { $ok = 0 }
        }
        $p = longueur($m)
        si ($e < $p) {
          $rs = $dg + ".0"
        } sinon {
          $rs = sous_chaine($m, 0, 1)
          si ($p > 1) { $rs = $rs + "." + sous_chaine($m, 1, $p) }
          $rs = $rs + "e+"
          si ($e < 10) { $rs = $rs + "0" }
          $rs = $rs + chaine($e)
        }
        si ($v < 0.0) { $rs = "-" + $rs }
      }
    }
  }
  retour $rs
}

# le signe d'un texte de littéral (les opérandes d'un pli sont toujours des
# littéraux — le « -0.0 » déjà plié porte son moins).
fn texte_neg($t: chaine) -> booleen {
  si (longueur($t) == 0) { retour faux }
  retour ordre($t[0]) == 45
}

# ══════════════════ émission RIR (passe 3) ══════════════════
# Miroir d'EmetteurRir : compteurs partagés — %t et suffixes de bloc comptent
# sans reprise sur tout le module. Les blocs sont déclarés dans l'ordre des
# push_back C++ ; le bloc en attente (test de tant_que, borne de pour) reçoit
# ses lignes avant même d'être listé, en portant déjà son nom : la liste $fb
# garde l'ordre de création et la fenêtre de lignes d'un bloc reste contiguë.
# Le curseur est toujours le dernier bloc déclaré — la coupure des « si »
# imbriqués tombe donc dans le « fin » intérieur, comme dans le moteur.

fn rir_of($b: chaine) -> chaine {
  si ($b == "entier") { retour "z" }
  si ($b == "rationnel") { retour "q" }
  si ($b == "flottant") { retour "f" }
  si ($b == "booleen") { retour "b" }
  si ($b == "chaine") { retour "s" }
  si ($b == "liste") { retour "l" }
  si ($b == "table") { retour "tbl" }
  retour "u"
}

fn tick_t($tc: liste<entier>) -> chaine {
  $s = "%t" + chaine(longueur($tc) - 1)
  joindre($tc, 0)
  retour $s
}

fn tick_b($bc: liste<entier>) -> chaine {
  $s = "_" + chaine(longueur($bc) - 1)
  joindre($bc, 0)
  retour $s
}

# type annoté d'un nœud : note de première visite, entrée de repli (littéral
# reconstitué du monde C++), littéral de source, puis le vide.
fn own($i: entier, $k: liste<chaine>, $tx: liste<chaine>, $tf: liste<chaine>,
       $foi: liste<entier>, $fot: liste<chaine>, $fof: liste<chaine>,
       $tni: liste<entier>, $tnt: liste<chaine>) -> chaine {
  $n = longueur($tni)
  $j = 1
  tant_que ($j < $n) {
    si ($tni[$j] == $i) { retour $tnt[$j] }
    $j = $j + 1
  }
  $vue = entree($i, $foi, $fot, $fof)
  si ($vue != "0") { retour t_entree(vue_flag($vue)) }
  si ($k[$i] == "lit") { retour t_from_texte($tx[$i], $tf[$i]) }
  retour "vide|vide"
}

fn e_bloc($fb: liste<chaine>, $cc: liste<entier>, $nom: chaine) {
  joindre($fb, $nom)
  joindre($cc, longueur($fb) - 1)
}

# $na : la ligne a un opérande — le dump C++ écrit « » + texte pour chaque
# opérande, même vide (une chaine vide se dump «cst.s » avec l'espace).
fn e_ligne($fl: liste<chaine>, $flb: liste<entier>, $fp: liste<chaine>, $cc: liste<entier>,
           $dest: chaine, $op: chaine, $argstr: chaine, $msg: chaine, $na: booleen) {
  $l = "  "
  si ($dest != "") { $l = $l + $dest + " = " }
  $l = $l + $op
  si ($na) { $l = $l + " " + $argstr }
  si ($msg != "") { $l = $l + " \"" + $msg + "\"" }
  joindre($fl, $l)
  joindre($flb, $cc[longueur($cc) - 1])
  joindre($fp, $op)
}

fn em_expr($i: entier, $ta: chaine,
           $k: liste<chaine>, $tx: liste<chaine>, $tf: liste<chaine>,
           $a: liste<entier>, $b: liste<entier>, $as: liste<entier>, $ac: liste<entier>,
           $lg: liste<entier>, $al: liste<entier>,
           $foi: liste<entier>, $fot: liste<chaine>, $fof: liste<chaine>,
           $tni: liste<entier>, $tnt: liste<chaine>,
           $tc: liste<entier>, $bc: liste<entier>,
           $fb: liste<chaine>, $fl: liste<chaine>, $flb: liste<entier>,
           $fp: liste<chaine>, $cc: liste<entier>) -> chaine {
  $ty = own($i, $k, $tx, $tf, $foi, $fot, $fof, $tni, $tnt)
  si ($ta == "u" et BASE($ty) != "vide") { $ta = rir_of(BASE($ty)) }
  $kd = $k[$i]
  $op0 = $tx[$i]
  $t = ""
  $g = ""
  $d = ""
  $opn = ""
  # — comparaison entre tampons : égalité générique, pas de conversion —
  si ($kd == "bin" et ($op0 == "==" ou $op0 == "!=")) {
    $tg = own($a[$i], $k, $tx, $tf, $foi, $fot, $fof, $tni, $tnt)
    $td = own($b[$i], $k, $tx, $tf, $foi, $fot, $fof, $tni, $tnt)
    si (BASE($tg) == "tampon" ou BASE($td) == "tampon") {
      $g = em_expr($a[$i], "u", $k, $tx, $tf, $a, $b, $as, $ac, $lg, $al, $foi, $fot, $fof, $tni, $tnt, $tc, $bc, $fb, $fl, $flb, $fp, $cc)
      $d = em_expr($b[$i], "u", $k, $tx, $tf, $a, $b, $as, $ac, $lg, $al, $foi, $fot, $fof, $tni, $tnt, $tc, $bc, $fb, $fl, $flb, $fp, $cc)
      $t = tick_t($tc)
      $opn = "cmp.eq.b"
      si ($op0 == "!=") { $opn = "cmp.ne.b" }
      e_ligne($fl, $flb, $fp, $cc, $t, $opn, $g + " " + $d, "", vrai)
      retour $t
    }
  }
  # — nœud replié = littéral du monde C++ : constante inline —
  $vue = entree($i, $foi, $fot, $fof)
  si ($vue != "0") {
    $t = tick_t($tc)
    e_ligne($fl, $flb, $fp, $cc, $t, "cst." + $ta, "@" + "c" + vue_texte($vue), "", vrai)
    retour $t
  }
  si ($kd == "lit") {
    $t = tick_t($tc)
    e_ligne($fl, $flb, $fp, $cc, $t, "cst." + $ta, "@" + "c" + $tx[$i], "", vrai)
    retour $t
  }
  si ($kd == "ch") {
    $t = tick_t($tc)
    e_ligne($fl, $flb, $fp, $cc, $t, "cst.s", $tx[$i], "", vrai)
    retour $t
  }
  si ($kd == "bol") {
    $t = tick_t($tc)
    $vv = "faux"
    si ($tx[$i] == "1") { $vv = "vrai" }
    e_ligne($fl, $flb, $fp, $cc, $t, "cst.b", $vv, "", vrai)
    retour $t
  }
  si ($kd == "v") { retour "%" + $tx[$i] }
  si ($kd == "idx") {
    $g = em_expr($a[$i], "u", $k, $tx, $tf, $a, $b, $as, $ac, $lg, $al, $foi, $fot, $fof, $tni, $tnt, $tc, $bc, $fb, $fl, $flb, $fp, $cc)
    $d = em_expr($b[$i], "z", $k, $tx, $tf, $a, $b, $as, $ac, $lg, $al, $foi, $fot, $fof, $tni, $tnt, $tc, $bc, $fb, $fl, $flb, $fp, $cc)
    $t = tick_t($tc)
    e_ligne($fl, $flb, $fp, $cc, $t, "index", $g + " " + $d, "", vrai)
    retour $t
  }
  si ($kd == "bin") {
    $g = em_expr($a[$i], "u", $k, $tx, $tf, $a, $b, $as, $ac, $lg, $al, $foi, $fot, $fof, $tni, $tnt, $tc, $bc, $fb, $fl, $flb, $fp, $cc)
    $d = em_expr($b[$i], "u", $k, $tx, $tf, $a, $b, $as, $ac, $lg, $al, $foi, $fot, $fof, $tni, $tnt, $tc, $bc, $fb, $fl, $flb, $fp, $cc)
    $t = tick_t($tc)
    si ($op0 == "+") { $opn = "add" }
    sinon si ($op0 == "-") { $opn = "sub" }
    sinon si ($op0 == "*") { $opn = "mul" }
    sinon si ($op0 == "/") { $opn = "div" }
    sinon si ($op0 == "div") { $opn = "divf" }
    sinon si ($op0 == "^") { $opn = "pow" }
    sinon si ($op0 == "%") { $opn = "rem.z" }
    sinon si ($op0 == "==") { $opn = "cmp.eq" }
    sinon si ($op0 == "!=") { $opn = "cmp.ne" }
    sinon si ($op0 == "<") { $opn = "cmp.lt" }
    sinon si ($op0 == "<=") { $opn = "cmp.le" }
    sinon si ($op0 == ">") { $opn = "cmp.gt" }
    sinon si ($op0 == ">=") { $opn = "cmp.ge" }
    sinon si ($op0 == "et") { $opn = "and" }
    sinon si ($op0 == "ou") { $opn = "or" }
    sinon { $opn = $op0 }
    si ($op0 == "≈" ou $op0 == "?=" ou $op0 == "=~") { $opn = "cmp.approx" }
    si ($opn != "cmp.approx") { $opn = $opn + "." + $ta }
    e_ligne($fl, $flb, $fp, $cc, $t, $opn, $g + " " + $d, "", vrai)
    retour $t
  }
  si ($kd == "uni") {
    $g = em_expr($a[$i], $ta, $k, $tx, $tf, $a, $b, $as, $ac, $lg, $al, $foi, $fot, $fof, $tni, $tnt, $tc, $bc, $fb, $fl, $flb, $fp, $cc)
    $t = tick_t($tc)
    $opn = "neg." + $ta
    si ($tx[$i] == "non") { $opn = "not.b" }
    e_ligne($fl, $flb, $fp, $cc, $t, $opn, $g, "", vrai)
    retour $t
  }
  si ($kd == "fac") {
    $g = em_expr($a[$i], "z", $k, $tx, $tf, $a, $b, $as, $ac, $lg, $al, $foi, $fot, $fof, $tni, $tnt, $tc, $bc, $fb, $fl, $flb, $fp, $cc)
    $t = tick_t($tc)
    e_ligne($fl, $flb, $fp, $cc, $t, "call @builtin.factorielle", $g, "", vrai)
    retour $t
  }
  si ($kd == "lst") {
    $t = tick_t($tc)
    e_ligne($fl, $flb, $fp, $cc, $t, "liste.new", "", "", faux)
    $sa = $as[$i]
    $nb = $ac[$i]
    $j = 0
    tant_que ($j < $nb) {
      $d = em_expr($al[$sa + $j], "u", $k, $tx, $tf, $a, $b, $as, $ac, $lg, $al, $foi, $fot, $fof, $tni, $tnt, $tc, $bc, $fb, $fl, $flb, $fp, $cc)
      e_ligne($fl, $flb, $fp, $cc, "", "liste.push", $t + " " + $d, "", vrai)
      $j = $j + 1
    }
    retour $t
  }
  si ($kd == "app") {
    $sa = $as[$i]
    $nb = $ac[$i]
    $args = ""
    $j = 0
    tant_que ($j < $nb) {
      $d = em_expr($al[$sa + $j], "u", $k, $tx, $tf, $a, $b, $as, $ac, $lg, $al, $foi, $fot, $fof, $tni, $tnt, $tc, $bc, $fb, $fl, $flb, $fp, $cc)
      si ($args == "") { $args = $d } sinon { $args = $args + " " + $d }
      $j = $j + 1
    }
    $t = tick_t($tc)
    $nom = $tx[$i]
    $nq = longueur($nom)
    $q = 0
    $okq = 1
    tant_que ($okq == 1) {
      si ($q >= $nq) { $okq = 0 }
      sinon si (ordre($nom[$q]) == 59) { $okq = 0 }
      sinon { $q = $q + 1 }
    }
    si ($q < $nq) { $nom = sous_chaine($nom, $q + 1, $nq) }
    $asg = "$" + $nom
    si ($args != "") { $asg = $asg + " " + $args }
    e_ligne($fl, $flb, $fp, $cc, $t, "call", $asg, "", vrai)
    retour $t
  }
  retour "%?"
}

fn em_instr($ix: entier,
            $k: liste<chaine>, $tx: liste<chaine>, $tf: liste<chaine>,
            $a: liste<entier>, $b: liste<entier>, $as: liste<entier>, $ac: liste<entier>,
            $lg: liste<entier>, $al: liste<entier>,
            $ik: liste<chaine>, $ino: liste<chaine>, $ie: liste<entier>, $ie2: liste<entier>,
            $ib1: liste<chaine>, $ib2: liste<chaine>, $ilg: liste<entier>,
            $foi: liste<entier>, $fot: liste<chaine>, $fof: liste<chaine>,
            $tni: liste<entier>, $tnt: liste<chaine>,
            $tri: liste<entier>, $trt: liste<chaine>,
            $tc: liste<entier>, $bc: liste<entier>,
            $fb: liste<chaine>, $fl: liste<chaine>, $flb: liste<entier>,
            $fp: liste<chaine>, $cc: liste<entier>) {
  $kd = $ik[$ix]
  $lgn = $ilg[$ix]
  $v = ""
  $sf = ""
  si ($kd == "aff") {
    $v = em_expr($ie[$ix], "u", $k, $tx, $tf, $a, $b, $as, $ac, $lg, $al, $foi, $fot, $fof, $tni, $tnt, $tc, $bc, $fb, $fl, $flb, $fp, $cc)
    e_ligne($fl, $flb, $fp, $cc, "", "afficher", $v, "", vrai)
    retour
  }
  si ($kd == "lire") {
    $t = tick_t($tc)
    $nu = "1"
    $nn = longueur($tri)
    $j = 1
    $trouve = 0
    tant_que ($j < $nn et $trouve == 0) {
      si ($tri[$j] == $ix) { $nu = $trt[$j]; $trouve = 1 }
      $j = $j + 1
    }
    e_ligne($fl, $flb, $fp, $cc, $t, "lire", "%" + $ino[$ix], $nu, vrai)
    e_ligne($fl, $flb, $fp, $cc, "", "affect", $t, $ino[$ix], vrai)
    retour
  }
  si ($kd == "affi") {
    $v = em_expr($ie[$ix], "u", $k, $tx, $tf, $a, $b, $as, $ac, $lg, $al, $foi, $fot, $fof, $tni, $tnt, $tc, $bc, $fb, $fl, $flb, $fp, $cc)
    e_ligne($fl, $flb, $fp, $cc, "", "affect", $v, $ino[$ix], vrai)
    retour
  }
  si ($kd == "si") {
    $v = em_expr($ie[$ix], "b", $k, $tx, $tf, $a, $b, $as, $ac, $lg, $al, $foi, $fot, $fof, $tni, $tnt, $tc, $bc, $fb, $fl, $flb, $fp, $cc)
    $sf = tick_b($bc)
    $l1 = ".alors_" + chaine($lgn) + $sf
    $l2 = ".sinon_" + chaine($lgn) + $sf
    $l3 = ".fin_" + chaine($lgn) + $sf
    e_ligne($fl, $flb, $fp, $cc, "", "br.cond", $v + " " + $l1 + " " + $l2, "", vrai)
    e_bloc($fb, $cc, $l1)
    em_corps($ib1[$ix], $k, $tx, $tf, $a, $b, $as, $ac, $lg, $al, $ik, $ino, $ie, $ie2, $ib1, $ib2, $ilg, $foi, $fot, $fof, $tni, $tnt, $tri, $trt, $tc, $bc, $fb, $fl, $flb, $fp, $cc)
    e_ligne($fl, $flb, $fp, $cc, "", "br", $l3, "", vrai)
    e_bloc($fb, $cc, $l2)
    si ($ib2[$ix] != "") {
      em_corps($ib2[$ix], $k, $tx, $tf, $a, $b, $as, $ac, $lg, $al, $ik, $ino, $ie, $ie2, $ib1, $ib2, $ilg, $foi, $fot, $fof, $tni, $tnt, $tri, $trt, $tc, $bc, $fb, $fl, $flb, $fp, $cc)
    }
    e_ligne($fl, $flb, $fp, $cc, "", "br", $l3, "", vrai)
    e_bloc($fb, $cc, $l3)
    retour
  }
  si ($kd == "tq") {
    $sf = tick_b($bc)
    $l1 = ".cond_" + chaine($lgn) + $sf
    $l2 = ".corps_" + chaine($lgn) + $sf
    $l3 = ".fin_" + chaine($lgn) + $sf
    e_ligne($fl, $flb, $fp, $cc, "", "br", $l1, "", vrai)
    e_bloc($fb, $cc, $l1)
    $v = em_expr($ie[$ix], "b", $k, $tx, $tf, $a, $b, $as, $ac, $lg, $al, $foi, $fot, $fof, $tni, $tnt, $tc, $bc, $fb, $fl, $flb, $fp, $cc)
    e_ligne($fl, $flb, $fp, $cc, "", "br.cond", $v + " " + $l2 + " " + $l3, "", vrai)
    e_bloc($fb, $cc, $l2)
    em_corps($ib1[$ix], $k, $tx, $tf, $a, $b, $as, $ac, $lg, $al, $ik, $ino, $ie, $ie2, $ib1, $ib2, $ilg, $foi, $fot, $fof, $tni, $tnt, $tri, $trt, $tc, $bc, $fb, $fl, $flb, $fp, $cc)
    e_ligne($fl, $flb, $fp, $cc, "", "br", $l1, "", vrai)
    e_bloc($fb, $cc, $l3)
    retour
  }
  si ($kd == "pour") {
    $var = $ino[$ix]
    $sf = tick_b($bc)
    $l1 = ".cond_" + chaine($lgn) + $sf
    $l2 = ".corps_" + chaine($lgn) + $sf
    $l3 = ".incr_" + chaine($lgn) + $sf
    $l4 = ".fin_" + chaine($lgn) + $sf
    $v = em_expr($ie[$ix], "z", $k, $tx, $tf, $a, $b, $as, $ac, $lg, $al, $foi, $fot, $fof, $tni, $tnt, $tc, $bc, $fb, $fl, $flb, $fp, $cc)
    e_ligne($fl, $flb, $fp, $cc, "", "affect", $v, $var, vrai)
    e_ligne($fl, $flb, $fp, $cc, "", "br", $l1, "", vrai)
    e_bloc($fb, $cc, $l1)
    $d = em_expr($ie2[$ix], "z", $k, $tx, $tf, $a, $b, $as, $ac, $lg, $al, $foi, $fot, $fof, $tni, $tnt, $tc, $bc, $fb, $fl, $flb, $fp, $cc)
    $ct = tick_t($tc)
    e_ligne($fl, $flb, $fp, $cc, $ct, "cmp.lt.z", "%" + $var + " " + $d, "", vrai)
    e_ligne($fl, $flb, $fp, $cc, "", "br.cond", $ct + " " + $l2 + " " + $l4, "", vrai)
    e_bloc($fb, $cc, $l2)
    em_corps($ib1[$ix], $k, $tx, $tf, $a, $b, $as, $ac, $lg, $al, $ik, $ino, $ie, $ie2, $ib1, $ib2, $ilg, $foi, $fot, $fof, $tni, $tnt, $tri, $trt, $tc, $bc, $fb, $fl, $flb, $fp, $cc)
    e_ligne($fl, $flb, $fp, $cc, "", "br", $l3, "", vrai)
    e_bloc($fb, $cc, $l3)
    $nt = tick_t($tc)
    e_ligne($fl, $flb, $fp, $cc, $nt, "add.z", "%" + $var + " @c1", "", vrai)
    e_ligne($fl, $flb, $fp, $cc, "", "affect", $nt, $var, vrai)
    e_ligne($fl, $flb, $fp, $cc, "", "br", $l1, "", vrai)
    e_bloc($fb, $cc, $l4)
    retour
  }
  si ($kd == "retour") {
    si ($ie[$ix] >= 0) {
      $v = em_expr($ie[$ix], "u", $k, $tx, $tf, $a, $b, $as, $ac, $lg, $al, $foi, $fot, $fof, $tni, $tnt, $tc, $bc, $fb, $fl, $flb, $fp, $cc)
      e_ligne($fl, $flb, $fp, $cc, "", "ret", $v, "", vrai)
    }
    sinon {
      e_ligne($fl, $flb, $fp, $cc, "", "ret.u", "", "", faux)
    }
    retour
  }
  si ($kd == "expr") {
    em_expr($ie[$ix], "u", $k, $tx, $tf, $a, $b, $as, $ac, $lg, $al, $foi, $fot, $fof, $tni, $tnt, $tc, $bc, $fb, $fl, $flb, $fp, $cc)
    retour
  }
  si ($kd == "bloc") {
    em_corps($ib1[$ix], $k, $tx, $tf, $a, $b, $as, $ac, $lg, $al, $ik, $ino, $ie, $ie2, $ib1, $ib2, $ilg, $foi, $fot, $fof, $tni, $tnt, $tri, $trt, $tc, $bc, $fb, $fl, $flb, $fp, $cc)
  }
}

fn em_corps($cs: chaine,
            $k: liste<chaine>, $tx: liste<chaine>, $tf: liste<chaine>,
            $a: liste<entier>, $b: liste<entier>, $as: liste<entier>, $ac: liste<entier>,
            $lg: liste<entier>, $al: liste<entier>,
            $ik: liste<chaine>, $ino: liste<chaine>, $ie: liste<entier>, $ie2: liste<entier>,
            $ib1: liste<chaine>, $ib2: liste<chaine>, $ilg: liste<entier>,
            $foi: liste<entier>, $fot: liste<chaine>, $fof: liste<chaine>,
            $tni: liste<entier>, $tnt: liste<chaine>,
            $tri: liste<entier>, $trt: liste<chaine>,
            $tc: liste<entier>, $bc: liste<entier>,
            $fb: liste<chaine>, $fl: liste<chaine>, $flb: liste<entier>,
            $fp: liste<chaine>, $cc: liste<entier>) {
  $j = 0
  $n = longueur($cs)
  $cur = ""
  tant_que ($j < $n) {
    $c = ordre($cs[$j])
    si ($c == 44) {
      si ($cur != "") {
        em_instr(entier($cur), $k, $tx, $tf, $a, $b, $as, $ac, $lg, $al, $ik, $ino, $ie, $ie2, $ib1, $ib2, $ilg, $foi, $fot, $fof, $tni, $tnt, $tri, $trt, $tc, $bc, $fb, $fl, $flb, $fp, $cc)
        $cur = ""
      }
    }
    sinon { $cur = $cur + $cs[$j] }
    $j = $j + 1
  }
  si ($cur != "") {
    em_instr(entier($cur), $k, $tx, $tf, $a, $b, $as, $ac, $lg, $al, $ik, $ino, $ie, $ie2, $ib1, $ib2, $ilg, $foi, $fot, $fof, $tni, $tnt, $tri, $trt, $tc, $bc, $fb, $fl, $flb, $fp, $cc)
  }
}

fn em_fonction($nom: chaine, $ps: chaine, $ra: chaine, $corps: chaine,
               $k: liste<chaine>, $tx: liste<chaine>, $tf: liste<chaine>,
               $a: liste<entier>, $b: liste<entier>, $as: liste<entier>, $ac: liste<entier>,
               $lg: liste<entier>, $al: liste<entier>,
               $ik: liste<chaine>, $ino: liste<chaine>, $ie: liste<entier>, $ie2: liste<entier>,
               $ib1: liste<chaine>, $ib2: liste<chaine>, $ilg: liste<entier>,
               $foi: liste<entier>, $fot: liste<chaine>, $fof: liste<chaine>,
               $tni: liste<entier>, $tnt: liste<chaine>,
               $tri: liste<entier>, $trt: liste<chaine>,
               $tc: liste<entier>, $bc: liste<entier>,
               $outl: liste<chaine>) {
  $fb = [""]
  $fl = [""]
  $flb = [0]
  $fp = [""]
  $cc = [0]
  e_bloc($fb, $cc, ".entree")
  em_corps($corps, $k, $tx, $tf, $a, $b, $as, $ac, $lg, $al, $ik, $ino, $ie, $ie2, $ib1, $ib2, $ilg, $foi, $fot, $fof, $tni, $tnt, $tri, $trt, $tc, $bc, $fb, $fl, $flb, $fp, $cc)
  # — ret. implicite quand la dernière ligne n'est pas un retour —
  $ret_ok = 0
  si (longueur($fp) > 1) {
    $pop = $fp[longueur($fp) - 1]
    si (longueur($pop) >= 3) {
      si (sous_chaine($pop, 0, 3) == "ret") { $ret_ok = 1 }
    }
  }
  si ($ret_ok == 0) { e_ligne($fl, $flb, $fp, $cc, "", "ret.u", "", "", faux) }
  # — assemblage : en-tête (paramètres sous la forme étonnante du dump,
  # nom %nom : type), puis chaque bloc avec sa fenêtre de lignes —
  joindre($outl, "")
  $tt = "fonction " + $nom + "("
  $np = nb_pieces($ps, 59)
  $u = 1
  tant_que ($u <= $np) {
    si ($u > 1) { $tt = $tt + ", " }
    $ent = piece($ps, 59, $u)
    $pp = premier_pipe($ent)
    $pn = sous_chaine($ent, 0, $pp)
    $pt = sous_chaine($ent, $pp + 1, longueur($ent))
    $tt = $tt + $pn + " %" + $pn + " : " + rir_of(BASE($pt))
    $u = $u + 1
  }
  $tt = $tt + ") -> " + rir_of(BASE($ra)) + " {"
  joindre($outl, $tt)
  $ln = 1
  $bk = 1
  tant_que ($bk < longueur($fb)) {
    joindre($outl, $fb[$bk] + ":")
    $arret = 0
    tant_que ($ln < longueur($fl) et $arret == 0) {
      si ($flb[$ln] != $bk) { $arret = 1 }
      sinon {
        joindre($outl, $fl[$ln])
        $ln = $ln + 1
      }
    }
    $bk = $bk + 1
  }
  joindre($outl, "}")
}
